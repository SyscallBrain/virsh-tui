fn ev(code: KeyCode, modifiers: KeyModifiers) -> KeyEvent {
    KeyEvent {
        code,
        modifiers,
        kind: KeyEventKind::Press,
        state: KeyEventState::empty(),
    }
}
fn plain(c: char) -> KeyEvent {
    ev(KeyCode::Char(c), KeyModifiers::empty())
}
fn shift(c: char) -> KeyEvent {
    ev(KeyCode::Char(c), KeyModifiers::SHIFT)
}
fn ctrl(c: char) -> KeyEvent {
    ev(KeyCode::Char(c), KeyModifiers::CONTROL)
}
fn job_argv(app: &App) -> Vec<Vec<String>> {
    app.active_job
        .as_ref()
        .map(|j| {
            j.plans
                .iter()
                .flat_map(|(_, p)| p.steps.iter().map(|s| s.argv.clone()))
                .collect()
        })
        .unwrap_or_default()
}

#[test]
fn p0_2_selection_hops_under_cpu_sort() {
    use crate::backend::RawDomainStats;
    use std::collections::HashMap;
    let mk = |name: &str| crate::model::DomainSummary {
        name: name.into(),
        uuid: String::new(),
        id: None,
        state: DomainState::Running,
        autostart: false,
        persistent: true,
        vcpus: 1,
        max_vcpus: 1,
        mem_kib: 1_048_576,
        max_mem_kib: 1_048_576,
        has_managed_save: false,
    };
    let summaries = vec![mk("a"), mk("b"), mk("c")];
    let stats = |cpu: [u64; 3]| -> Vec<RawDomainStats> {
        ["a", "b", "c"]
            .iter()
            .zip(cpu)
            .map(|(n, c)| {
                let mut v = HashMap::new();
                v.insert("state.state".to_string(), "1".to_string());
                v.insert("vcpu.current".to_string(), "1".to_string());
                v.insert("cpu.time".to_string(), c.to_string());
                RawDomainStats {
                    domain: n.to_string(),
                    values: v,
                }
            })
            .collect()
    };
    let mut dash = ui::views::dashboard::DashboardState::from_summaries(&summaries);
    dash.sort = ui::views::dashboard::Sort::Cpu;
    let t0 = std::time::Instant::now();
    let mut sample = |dash: &mut ui::views::dashboard::DashboardState, cpu: [u64; 3], secs: u64| {
        apply_live_msg(
            "test:///default",
            dash,
            LiveMsg::Sample {
                summaries: summaries.clone(),
                stats: stats(cpu),
                proc_stat: None,
                uptimes: HashMap::new(),
                ifaces: HashMap::new(),
                at: t0 + Duration::from_secs(secs),
            },
        );
    };
    sample(&mut dash, [0, 0, 0], 0);
    sample(&mut dash, [100_000_000, 500_000_000, 900_000_000], 1);
    dash.select_name("a");
    let before = dash.selected().name.clone();
    sample(&mut dash, [200_000_000, 1_000_000_000, 1_800_000_000], 2);
    let after = dash.selected().name.clone();
    eprintln!("PROBE p0_2: before={before} after={after}");
    assert_eq!(before, "a");
    assert_ne!(after, "a");
}

fn draw_at(app: &App, w: u16, h: u16) -> Result<(), String> {
    let res = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
        let backend = ratatui::backend::TestBackend::new(w, h);
        let mut terminal = ratatui::Terminal::new(backend).unwrap();
        terminal.draw(|f| app.draw(f)).unwrap();
    }));
    res.map_err(|e| {
        e.downcast_ref::<String>()
            .cloned()
            .or_else(|| e.downcast_ref::<&str>().map(|s| s.to_string()))
            .unwrap_or_default()
    })
}

        ("whichkey", Box::new(|| { let mut a = fixture_app(); a.engine.feed(&crate::input::key::KeySeq::parse("Space")); a.which_key_since = Some(std::time::Instant::now() - Duration::from_secs(1)); a })),

        assert!(!plans.iter().any(|(t, _)| *t == p));
    }
}

#[tokio::test]
async fn r2_set_does_not_update_runtime() {
    let mut app = fixture_app();
    crate::config::set_runtime(&app.config);
    app.ex_line = Some(crate::input::textinput::TextInput::with("set viewer=remote-viewer"));
    app.handle_key_event(ev(KeyCode::Enter, KeyModifiers::empty()));
    let rt = crate::config::runtime().console.viewer;
    eprintln!("PROBE r2_set: message={:?} config={} runtime={rt}", app.dash.message, app.config.console.viewer);
    assert_eq!(app.config.console.viewer, "remote-viewer");
    assert_eq!(rt, "virt-viewer");
}
